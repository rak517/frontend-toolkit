# 시작하기

## 설치

```bash
# npm
npm install @frontend-toolkit/hooks @frontend-toolkit/components

# pnpm
pnpm add @frontend-toolkit/hooks @frontend-toolkit/components

# yarn
yarn add @frontend-toolkit/hooks @frontend-toolkit/components
```

## 사용법

### Hooks

```tsx
import { useDebounce } from '@frontend-toolkit/hooks';

function SearchInput() {
  const [search, setSearch] = useState('');
  const debouncedSearch = useDebounce(search, 300);

  useEffect(() => {
    // API 호출
    fetchResults(debouncedSearch);
  }, [debouncedSearch]);

  return <input value={search} onChange={e => setSearch(e.target.value)} />;
}
```

### Components

```tsx
import { InViewTrigger } from '@frontend-toolkit/components';

function LazySection() {
  return (
    <InViewTrigger onEnter={() => console.log('화면에 보임!')}>
      <div>내용</div>
    </InViewTrigger>
  );
}
```
